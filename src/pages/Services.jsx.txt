/**
 * Services.jsx — Route: /services
 * 8 service cards in a responsive CSS Grid.
 * Data mapped entirely from data.js.
 */

import { Link } from 'react-router-dom';
import { useState, useEffect } from 'react';
import { client, urlFor } from '../sanity';
import { useLanguage } from '../context/LanguageContext';
import { services } from '../data';
import GradientCard from '../components/GradientCard';
import SectionWrapper from '../components/SectionWrapper';
import PageHeading from '../components/PageHeading';

// ── Individual service card ───────────────────────────────────────────────────
function ServiceCard({ service }) {
  const { t } = useLanguage();

  return (
    <GradientCard as="article" aria-label={t(service.titleEn, service.titleAr)}>
      <div className="flex flex-col gap-4 p-8">

        {/* Icon */}
        <div className="flex h-14 w-14 items-center justify-center rounded-xl border border-slate-200 bg-blue-500/10 text-2xl transition-all duration-300 group-hover:border-blue-500/50 group-hover:bg-blue-500/20 dark:border-blue-500/20 overflow-hidden p-2">
          {service.icon ? (
            <img src={urlFor(service.icon).url()} alt="Icon" className="w-full h-full object-contain" />
          ) : null}
        </div>
        
        {/* Title */}
        <h3 className="text-xl font-bold leading-snug text-slate-900 dark:text-white">
          {t(service.titleEn, service.titleAr)}
        </h3>

        {/* Description */}
        <p className="text-sm leading-relaxed text-slate-600 dark:text-slate-400">
          {t(service.descEn, service.descAr)}
        </p>
      </div>
    </GradientCard>
  );
}

// ─────────────────────────────────────────────────────────────────────────────
// SERVICES PAGE
// ─────────────────────────────────────────────────────────────────────────────
export default function Services() {
  const { t } = useLanguage();
  
  const [servicesData, setServicesData] = useState([]);
  const [isLoading, setIsLoading] = useState(true);

  useEffect(() => {
    client.fetch('*[_type == "service"]')
      .then((data) => {
        setServicesData(data);
        setIsLoading(false);
      })
      .catch((err) => {
        console.error("Error fetching services:", err);
        setIsLoading(false);
      });
  }, []);

  return (
    <>
      {/* ═══════════════════════════════════════════════════════
          SERVICES GRID
          ═══════════════════════════════════════════════════════ */}
      <SectionWrapper>
        {/* Orbs */}
        <div className="orb -start-40 top-0 h-[500px] w-[500px] bg-blue-600/15" />
        <div className="orb -end-20 bottom-20 h-[400px] w-[400px] bg-rose-600/10" />

        <div className="relative">
          <PageHeading
            badge={t('What I Offer', 'ماذا أقدم')}
            title={t('Services I Provide', 'الخدمات التي أقدمها')}
            subtitle={t(
              'I offer design and development services focused on building fast, modern, and user-friendly digital experiences.',
              'أقدم خدمات التصميم والتطوير مع التركيز على بناء تجارب رقمية سريعة وحديثة وسهلة الاستخدام.'
            )}
          />

         {/* auto-fit grid — adapts to any screen without media queries */}
          {isLoading ? (
            <div className="text-center py-10 text-slate-500 dark:text-slate-400">
              {t('Loading services...', 'جاري تحميل الخدمات...')}
            </div>
          ) : (
            <div
              className="grid gap-6"
              style={{ gridTemplateColumns: 'repeat(auto-fit, minmax(260px, 1fr))' }}
              role="list"
            >
              {servicesData.map((svc) => (
                <div key={svc._id} role="listitem" className="group">
                  <ServiceCard service={svc} />
                </div>
              ))}
            </div>
          )}
        </div>
      </SectionWrapper>

      {/* ═══════════════════════════════════════════════════════
          PROCESS SECTION
          ═══════════════════════════════════════════════════════ */}
      <SectionWrapper>
        <PageHeading
          badge={t('How I Work', 'كيف أعمل')}
          title={t('My Development Process', 'عملية التطوير')}
          subtitle={t(
            'A structured, collaborative approach to delivering high-quality results on time.',
            'نهج منظم وتعاوني لتقديم نتائج عالية الجودة في الوقت المحدد.'
          )}
        />

        <div className="grid gap-8 sm:grid-cols-2 lg:grid-cols-4">
          {[
            {
              step: '01',
              titleEn: 'Discovery',
              titleAr: 'الاستكشاف',
              descEn: 'Understanding your goals, requirements, and target audience.',
              descAr: 'فهم أهدافك ومتطلباتك والجمهور المستهدف.',
            },
            {
              step: '02',
              titleEn: 'Planning',
              titleAr: 'التخطيط',
              descEn: 'Defining the architecture, tech stack, and project timeline.',
              descAr: 'تحديد البنية والتقنيات والجدول الزمني للمشروع.',
            },
            {
              step: '03',
              titleEn: 'Development',
              titleAr: 'التطوير',
              descEn: 'Writing clean, modular code with regular progress updates.',
              descAr: 'كتابة كود نظيف ومنظم مع تحديثات منتظمة للتقدم.',
            },
            {
              step: '04',
              titleEn: 'Delivery',
              titleAr: 'التسليم',
              descEn: 'Testing, deployment, and post-launch support.',
              descAr: 'الاختبار والنشر ودعم ما بعد الإطلاق.',
            },
          ].map((item) => (
            <div key={item.step} className="flex flex-col gap-4">
              <span className="text-5xl font-black gradient-text opacity-60">
                {item.step}
              </span>
              <h3 className="text-lg font-bold text-white">
                {t(item.titleEn, item.titleAr)}
              </h3>
              <p className="text-sm leading-relaxed text-slate-400">
                {t(item.descEn, item.descAr)}
              </p>
            </div>
          ))}
        </div>
      </SectionWrapper>

      {/* ═══════════════════════════════════════════════════════
          CTA
          ═══════════════════════════════════════════════════════ */}
      <SectionWrapper>
        <div className="relative flex flex-col items-center gap-6 text-center">
          <div className="orb left-1/2 top-0 h-[300px] w-[500px] -translate-x-1/2 bg-blue-600/10" />
          <h2 className="text-3xl font-extrabold text-white md:text-4xl">
            {t('Ready to start your project?', 'هل أنت مستعد لبدء مشروعك؟')}
          </h2>
          <p className="max-w-lg text-base text-slate-400">
            {t(
              "Let's discuss your requirements and create something exceptional together.",
              'دعنا نناقش متطلباتك ونبني شيئاً استثنائياً معاً.'
            )}
          </p>
          <Link
            to="/contact"
            className="inline-flex items-center gap-2 rounded-full bg-gradient-to-r from-blue-500 to-rose-500 px-10 py-4 text-sm font-bold text-white shadow-glow-blue transition-all hover:opacity-90 active:scale-95"
          >
            {t('Start a Project', 'ابدأ مشروعاً')}
          </Link>
        </div>
      </SectionWrapper>
    </>
  );
}
